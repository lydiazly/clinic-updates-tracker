#----------------------------------------------------------------------|
# Build

# [Debian-based (Linux 6.10.14-linuxkit), python3.12, multi-stage]
tag='test'
tag='run'
tag='user'
docker build --target builder -t clinic-test:builder -f docker/Dockerfile .  # 351MB
docker build --target base -t clinic-test:base -f docker/Dockerfile .  # 314MB
docker build --target test -t clinic-test:test -f docker/Dockerfile .  # 314MB run:✓
docker build --target run -t clinic-test:run -f docker/Dockerfile .  # 949MB run:✓
docker build --target user -t clinic-test:user -f docker/Dockerfile .  # 949MB run:✓

# Run
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -e TEST_MODE=false -e SEND_EMAILS=false --rm clinic-test:$tag

# Other tests
tag='test'
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -u runner --rm clinic-test:$tag  # ✓

tag='run'
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -v "$(pwd)"/output:/app/output \
  -e TEST_MODE=true -e DAYS_BACK=3 -e SEND_EMAILS=true --rm clinic-test:$tag  # ✓
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -v "$(pwd)"/output:/app/output \
  -e TZ=UTC -e DAYS_BACK=1 -e MAX_ITEMS=10 --rm clinic-test:$tag  # ✓
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  --entrypoint pytest --rm clinic-test:$tag  # ✓

tag='user'
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  --network clinictracker_default --rm clinic-test:$tag list  # if db is running ✓
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -e PGSERVICE=prod --rm clinic-test:$tag list  # ✓
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -e PGSERVICE=prod --rm clinic-test:$tag --ignore-hash  # ✓
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -e PGSERVICE=prod --rm clinic-test:$tag --ignore-hash --send  # ✓

# Compose

# Build
docker compose build run-task
docker compose build user

# Database in docker
# use ./.pg_password as secrets
docker compose up -d db

# Stop and remove
docker compose down -v db

# Run (keep running if -e KEEP_ALIVE=true)
# Already set -it in test-mode
docker compose run --name clinic-test --rm test-mode  # ✓
docker compose run --name clinic-test -u root --rm test-mode  # ✓
docker compose run --name clinic-test -e DAYS_BACK=1 --rm test-mode ./docker/docker-entrypoint.sh # ✓
docker compose run --name clinic-test -e DAYS_BACK=1 \
  --entrypoint /app/docker/docker-entrypoint.sh --rm test-mode # ✓
docker compose run --name clinic-test -e SEND_EMAILS=true --rm run-task  # ✓
# Use db
docker compose run --name clinic-test --rm user-dev  # ✓
docker compose run --name clinic-test --rm user-dev list  # ✓
# Use external database
docker compose run --name clinic-test -e IGNORE_HASH=true --rm user list  # ✓

# Create & start container
docker compose up -d test-mode  # ✓
# Rebuild images before starting container
docker compose up --build -d test-mode
# Force rebuild without cache
docker compose build --no-cache test-mode
docker compose up -d test-mode

# Stop & remove containers and networks
docker compose down

# View and follow logs (running)
# docker compose logs -f test-mode

# Test (running)
docker compose run --name clinic-test -e KEEP_ALIVE=true -d test-mode  # ✓
docker compose exec -e DAYS_BACK=1 test-mode ./docker/docker-entrypoint.sh  # ✓
# Force stop & remove manually
docker compose kill test-mode
docker compose down --remove-orphans
