# Build

# [Debian-based (Linux 6.10.14-linuxkit), python3.12, multi-stage]
tag='test-pip'
tag='run-pip'
tag='user-pip'
docker build --target builder -t clinic-test:builder-pip -f docker/pip/Dockerfile .  # 330MB
docker build --target base -t clinic-test:base-pip -f docker/pip/Dockerfile .  # 336MB
docker build --target test -t clinic-test:test-pip -f docker/pip/Dockerfile .  # 336MB run:✓
docker build --target run -t clinic-test:run-pip -f docker/pip/Dockerfile .  # 971MB run:✓
docker build --target user -t clinic-test:user-pip -f docker/pip/Dockerfile .  # 971MB run:✓

# Run
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -e TEST_MODE=false -e SEND_EMAILS=false --rm clinic-test:$tag

# Other tests
tag='test-pip'
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -u runner --rm clinic-test:$tag  # ✓

tag='run-pip'
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -v "$(pwd)"/output:/app/output \
  -e TEST_MODE=true -e DAYS_BACK=3 -e SEND_EMAILS=true --rm clinic-test:$tag  # ✓
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  -v "$(pwd)"/output:/app/output \
  -e TZ=UTC -e DAYS_BACK=1 -e MAX_ITEMS=10 --rm clinic-test:$tag  # ✓
docker run -it --name clinic-test --env-file .env --env-file .secrets \
  --rm clinic-test:$tag pytest  # ✓

tag='user-pip'
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \

docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -e PGSERVICE=prod --rm clinic-test:$tag list  # ✓
docker run -it --name clinic-test --env-file .env.test --env-file .secrets \
  -e PGSERVICE=prod --rm clinic-test:$tag --forget-last  # ✓
